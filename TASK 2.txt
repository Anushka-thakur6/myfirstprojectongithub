    <ul>
        <li><strong>Email Phishing</strong> - Fake bank/PayPal emails</li>
        <li><strong>Spear Phishing</strong> - Targeted at you specifically</li>
        <li><strong>Smishing</strong> - SMS text messages</li>
        <li><strong>Vishing</strong> - Phone calls pretending to be tech support</li>
    </ul>

<div class="section">
    <h2> SECTION 2: Spot Phishing Emails & Fake Sites</h2>
    
    <div class="warning">
        <h4> RED FLAGS:</h4>
        <ul>
            <li><strong>Wrong sender:</strong> support@arnazon.com (not amazon.com)</li>
            <li><strong>Generic greeting:</strong> "Dear Customer" (not your name)</li>
            <li><strong>URGENT language:</strong> "Account closes in 24hrs!"</li>
            <li><strong>Typos/domains:</strong> paypa1.com, g00gle.com</li>
            <li><strong>Shortened URLs:</strong> Hover to see real destination</li>
        </ul>
    </div>
    
    <div class="safe">
        <h4> SAFE CHECKS:</h4>
        <ul>
            <li>Hover links (don't click)</li>
            <li>Check padlock/SSL is real</li>
            <li>Verify sender by phone</li>
        </ul>
    </div>
</div>

<script>
let score = 0;
function checkAnswer(question, isLegit) {
    const result = document.getElementById(`result${question}`);
    if (!isLegit) {
        score++;
        result.innerHTML = '<span style="color:green; font-weight:bold;"> CORRECT! Phishing detected!</span>';
    } else {
        result.innerHTML = '<span style="color:red; font-weight:bold;"> WRONG! This is phishing!</span>';
    }
    
    if (question === 2 && score === 1) {
        setTimeout(() => {
            alert(` Score: ${score}/2\nGreat job spotting phishing!\n\n Remember: Always verify suspicious emails!`);
        }, 500);
    }
}
</script>
